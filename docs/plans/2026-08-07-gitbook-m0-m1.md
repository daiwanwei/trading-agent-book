**Architecture:** 獨立 repo，`book/` 為 GitBook root。手寫敘事章節 + 兩個 Python 腳本：`generate_appendix.py`（從上游 repo 的 `skills-index.yaml` 與 `workflows/*.yaml` 生成附錄 A/B）、`check_book.py`（驗證 SUMMARY 與檔案一致、上游連結有效）。腳本走 TDD；章節寫作以固定五段模板 + 驗證腳本把關。

**Tech Stack:** Markdown + Mermaid（GitBook 原生支援）、Python 3.10+、PyYAML、pytest、gh CLI、GitBook.com Git Sync。

**Spec:** `docs/specs/2026-08-07-trading-agent-gitbook-design.md`（本計畫只覆蓋 M0+M1；M2–M5 之後另開計畫）

## Global Constraints

- 語言：繁體中文敘事；術語、指令、檔名、artifact 名稱保留英文原文。
- 上游 repo：`https://github.com/tradermonty/claude-trading-skills`，本機 clone 預設路徑 `~/Projects/wade/math/claude-trading-skills`。
- 重述不搬運：不整段翻譯上游英文素材；引用一律連結回上游 GitHub 具體檔案（格式 `https://github.com/tradermonty/claude-trading-skills/blob/main/<path>` 或 `/tree/main/<path>`）。
- 場景不虛構：場景只能使用真實 workflow 步驟、`decision_gate`、artifact 名稱（以上游 `workflows/*.yaml` 為準）。

- 立場：不提供投資建議；沿用上游「輸出是姿態不是訊號、不自動下單」的原則，每個實作章結尾重申。

```python
"""Generate appendix A (skills) and B (workflows) from the upstream repo."""

import argparse
import subprocess
import sys
from pathlib import Path

import yaml

DEFAULT_UPSTREAM = Path.home() / "Projects/wade/math/claude-trading-skills"
UPSTREAM_URL = "https://github.com/tradermonty/claude-trading-skills"


def load_skills_index(upstream: Path) -> dict:
    with open(upstream / "skills-index.yaml", encoding="utf-8") as f:
        return yaml.safe_load(f)


def load_workflows(upstream: Path) -> list:
    flows = []
    for path in sorted((upstream / "workflows").glob("*.yaml")):
        with open(path, encoding="utf-8") as f:
            flows.append(yaml.safe_load(f))
    return sorted(flows, key=lambda w: w["id"])


def get_upstream_commit(upstream: Path) -> str:
    try:
        result = subprocess.run(
            ["git", "-C", str(upstream), "rev-parse", "--short", "HEAD"],
            capture_output=True,
            text=True,
            check=True,
        )
        return result.stdout.strip()
    except (subprocess.CalledProcessError, FileNotFoundError):
        return "unknown"
```

```python
REQ_LABEL = {"required": "必需", "recommended": "建議", "optional": "可選"}


def _api_cell(skill: dict) -> str:
    parts = []
    for integ in skill.get("integrations", []):
        label = REQ_LABEL.get(integ.get("requirement"))
        if label:
            parts.append(f"{integ['id']}（{label}）")
    return "、".join(parts) if parts else "—"


def render_skills_appendix(index: dict, upstream_commit: str) -> str:
    lines = [
        "<!-- generated: true -->",
        f"<!-- source: tradermonty/claude-trading-skills@{upstream_commit} -->",
        "# 附錄 A：Skills 速查表",
        "",
        f"共 {len(index['skills'])} 個 skills，依分類排列。"
        "本頁由 `scripts/generate_appendix.py` 自動生成，請勿手動編輯。",
    ]
    for category in index["categories"]:
        cat_skills = [s for s in index["skills"] if s["category"] == category]
        if not cat_skills:
            continue
        lines += [
            "",
            f"## {category}",
            "",
            "| Skill | 摘要 | 節奏 | 難度 | API |",
            "|---|---|---|---|---|",
        ]
        for skill in cat_skills:
            link = f"[{skill['display_name']}]({UPSTREAM_URL}/tree/main/skills/{skill['id']})"
            summary = " ".join(str(skill.get("summary", "")).split())
            lines.append(
                f"| {link} | {summary} | {skill.get('timeframe', '—')} "
                f"| {skill.get('difficulty', '—')} | {_api_cell(skill)} |"
            )
    return "\n".join(lines) + "\n"
```

```python
            )
    return "\n".join(lines) + "\n"
```

`scripts/check_book.py`:

```python
"""Validate book structure: SUMMARY <-> files, upstream link targets."""

import argparse
import re
import sys
from pathlib import Path

DEFAULT_UPSTREAM = Path.home() / "Projects/wade/math/claude-trading-skills"
SUMMARY_LINK_RE = re.compile(r"\[[^\]]*\]\(([^)#]+\.md)\)")
UPSTREAM_LINK_RE = re.compile(
    r"https://github\.com/tradermonty/claude-trading-skills/(?:tree|blob)/main/([^)\s#]+)"
)


def parse_summary(text: str) -> list:
    return SUMMARY_LINK_RE.findall(text)


def check_summary_files(book_dir: Path) -> list:
    issues = []
    targets = parse_summary((book_dir / "SUMMARY.md").read_text(encoding="utf-8"))
    for target in targets:
        if not (book_dir / target).exists():
            issues.append(f"SUMMARY links to missing file: {target}")
    listed = set(targets)
    for md in book_dir.rglob("*.md"):
        rel = md.relative_to(book_dir).as_posix()
        if rel != "SUMMARY.md" and rel not in listed:
            issues.append(f"file not listed in SUMMARY: {rel}")
    return issues


def find_upstream_links(text: str) -> list:
    return UPSTREAM_LINK_RE.findall(text)


def check_upstream_links(book_dir: Path, upstream: Path) -> list:
    issues = []
    for md in sorted(book_dir.rglob("*.md")):
        for rel in find_upstream_links(md.read_text(encoding="utf-8")):
            if not (upstream / rel).exists():
                issues.append(
                    f"{md.relative_to(book_dir).as_posix()}: upstream path missing: {rel}"
                )
    return issues


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Validate book structure and links")
    parser.add_argument(
        "--book",
        type=Path,
        default=Path(__file__).resolve().parents[1] / "book",
    )
    parser.add_argument("--upstream", type=Path, default=DEFAULT_UPSTREAM)
    args = parser.parse_args(argv)

    issues = check_summary_files(args.book)
    if args.upstream.exists():
        issues += check_upstream_links(args.book, args.upstream)
    else:
        print(f"warning: upstream not found at {args.upstream}; link check skipped")

    for issue in issues:
        print(issue)
    print(f"{len(issues)} issue(s)")
    return 1 if issues else 0


if __name__ == "__main__":
    sys.exit(main())
```
